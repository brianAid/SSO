📘 Panduan SSO: React Client + Laravel Resource Server

React Client

📌 Alur Autentikasi:

  1. Redirect pengguna ke SSO Server
  2. Tangkap authorization code di callback URL
  3. Tukar code dengan access token dan refresh token
  4. Gunakan token untuk mengakses API Resource Server
  5. Refresh token secara berkala

1️⃣ Redirect ke login SSO Server:

const CLIENT_ID = "app1";
const REDIRECT_URI = encodeURIComponent("http://localhost:8000/callback");
const STATE = crypto.randomUUID(); // Generate secure state

// Redirect ke endpoint authorize SSO
window.location.href = 
  `http://localhost:3000/authorize?client_id=${CLIENT_ID}&redirect_uri=${REDIRECT_URI}&state=${STATE}`;

💡 Parameter Penting:

  • client_id: Identifikasi aplikasi client
  • redirect_uri: URL callback setelah login
  • state: Parameter keamanan untuk mencegah CSRF

2️⃣ Tangkap code dan tukar dengan token:

import { useEffect } from 'react';
import { useNavigate } from 'react-router-dom';

function CallbackPage() {
  const navigate = useNavigate();

  useEffect(() => {
    const params = new URLSearchParams(window.location.search);
    const code = params.get('code');
    const state = params.get('state');
    
    // Validasi state
    if (state !== localStorage.getItem('oauth_state')) {
      navigate('/error?message=invalid_state');
      return;
    }

    if (!code) {
      navigate('/error?message=no_code');
      return;
    }

    // Tukar code dengan token
    fetch("http://localhost:3000/token", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        code,
        client_id: "app1",
        client_secret: "secretapp1",
        redirect_uri: "http://localhost:8000/callback",
      }),
    })
      .then(res => res.json())
      .then(data => {
        if (data.error) {
          throw new Error(data.error);
        }
        
        // Simpan token
        localStorage.setItem("access_token", data.access_token);
        localStorage.setItem("refresh_token", data.refresh_token);
        navigate('/dashboard');
      })
      .catch(error => {
        console.error('Token exchange failed:', error);
        navigate('/error?message=auth_failed');
      });
  }, [navigate]);

  return <div>Loading...</div>;
}

3️⃣ Gunakan token untuk akses resource:

async function fetchProtectedData() {
  const token = localStorage.getItem("access_token");
  
  try {
    const response = await fetch("http://localhost:8001/api/profile", {
      headers: {
        Authorization: `Bearer ${token}`,
        'Content-Type': 'application/json'
      }
    });
    
    if (response.status === 401) {
      // Token expired, coba refresh
      await refreshToken();
      return fetchProtectedData();
    }
    
    if (!response.ok) {
      throw new Error('Failed to fetch data');
    }
    
    return await response.json();
  } catch (error) {
    console.error('API request failed:', error);
    throw error;
  }
}

4️⃣ Implementasi refresh token:

async function refreshToken() {
  const refreshToken = localStorage.getItem("refresh_token");
  if (!refreshToken) {
    throw new Error('No refresh token available');
  }

  try {
    const response = await fetch("http://localhost:3000/refresh", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        refresh_token: refreshToken,
        client_id: "app1",
        client_secret: "secretapp1"
      })
    });

    const data = await response.json();
    if (!response.ok) {
      throw new Error(data.error || 'Refresh failed');
    }

    localStorage.setItem("access_token", data.access_token);
    localStorage.setItem("refresh_token", data.refresh_token);
    return data.access_token;
  } catch (error) {
    console.error('Refresh token failed:', error);
    // Force logout
    localStorage.removeItem("access_token");
    localStorage.removeItem("refresh_token");
    window.location.href = '/login';
    throw error;
  }
}

// Setup interval untuk auto refresh
useEffect(() => {
  const interval = setInterval(async () => {
    try {
      await refreshToken();
    } catch (error) {
      clearInterval(interval);
    }
  }, 9 * 60 * 1000); // Refresh 1 menit sebelum expired (asumsi token expired 10 menit)

  return () => clearInterval(interval);
}, []);

🔒 Best Practices:

  • Gunakan httpOnly cookies untuk production
  • Implementasi CSRF protection dengan state parameter
  • Refresh token sebelum expired (9 menit untuk token 10 menit)
  • Handle error dengan baik (invalid token, network issues)

Laravel Resource Server

🔐 Fungsi Resource Server:

  • Verifikasi JWT menggunakan JWKS dari SSO Server
  • Mencocokkan user lokal berdasarkan username
  • Mengontrol akses berdasarkan role
  • Menyediakan API yang dilindungi

1️⃣ Install package JWT:

composer require firebase/php-jwt

2️⃣ Middleware verifyToken:

<?php

namespace App\Http\Middleware;

use Closure;
use Firebase\JWT\JWT;
use Firebase\JWT\JWK;
use Illuminate\Http\Request;
use App\Models\User;
use Illuminate\Support\Facades\Log;

class verifyToken
{
    public function handle(Request $request, Closure $next)
    {
        $authHeader = $request->header('Authorization');
        
        if (!$authHeader || !str_starts_with($authHeader, 'Bearer ')) {
            return response()->json([
                'error' => 'unauthorized',
                'message' => 'Missing or invalid Authorization header'
            ], 401);
        }

        $token = substr($authHeader, 7);
        
        try {
            // Dapatkan public keys dari SSO Server
            $jwks = json_decode(file_get_contents('http://localhost:3000/.well-known/jwks.json'), true);
            $keys = JWK::parseKeySet($jwks);
            
            // Verifikasi dan decode token
            $decoded = JWT::decode($token, $keys);
            
            // Cari user berdasarkan username
            $user = User::with('role')
                ->where('username', $decoded->username)
                ->first();
            
            if (!$user) {
                return response()->json([
                    'error' => 'user_not_found',
                    'message' => 'User not found in local database'
                ], 404);
            }

            // Tambahkan user ke request
            $request->merge([
                'auth_user' => $user,
                'auth_role' => $user->role->name,
                'auth_token' => $decoded
            ]);

        } catch (\Throwable $e) {
            Log::error('JWT verification failed: ' . $e->getMessage());
            
            return response()->json([
                'error' => 'invalid_token',
                'message' => 'Token verification failed'
            ], 401);
        }

        return $next($request);
    }
}

3️⃣ Middleware CheckRole:

<?php

namespace App\Http\Middleware;

use Closure;

class CheckRole
{
    public function handle($request, Closure $next, $role)
    {
        if ($request->auth_role !== $role) {
            return response()->json([
                'error' => 'forbidden',
                'message' => 'Insufficient permissions'
            ], 403);
        }
        
        return $next($request);
    }
}

4️⃣ Daftarkan middleware (Laravel 11):

// bootstrap/app.php

$app->routeMiddleware([
    'auth.sso' => \App\Http\Middleware\verifyToken::class,
    'role' => \App\Http\Middleware\CheckRole::class,
]);

5️⃣ Implementasi route:

use Illuminate\Support\Facades\Route;
use Illuminate\Http\Request;

// Grup route yang membutuhkan autentikasi
Route::middleware(['auth.sso'])->group(function () {
    
    // Endpoint profile untuk semua user terautentikasi
    Route::get('/api/profile', function (Request $request) {
        return response()->json([
            'user' => [
                'id' => $request->auth_user->id,
                'username' => $request->auth_user->username,
                'email' => $request->auth_user->email
            ],
            'role' => $request->auth_role,
            'token_exp' => $request->auth_token->exp
        ]);
    });

    // Endpoint khusus admin
    Route::prefix('/api/admin')->middleware('role:admin')->group(function () {
        Route::get('/dashboard', function () {
            return response()->json(['message' => 'Admin dashboard data']);
        });
        
        Route::get('/users', 'AdminController@getUsers');
    });

    // Endpoint khusus user
    Route::prefix('/api/user')->middleware('role:user')->group(function () {
        Route::get('/profile', 'UserController@getProfile');
    });
});

🗂️ Struktur Database:

users
- id
- username (unik, harus sama dengan di SSO Server)
- email
- password (optional, tergantung implementasi)
- role_id
- created_at
- updated_at

roles
- id
- name (admin, user, manager, etc.)
- permissions (json)
- created_at
- updated_at

user_clients (optional, untuk multi-client)
- user_id
- client_id
- created_at
